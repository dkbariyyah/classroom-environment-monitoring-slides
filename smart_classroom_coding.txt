#include <WiFi.h>
#include <PubSubClient.h>
#include "DHT.h"

// --- Hardware Pin Configurations ---
#define DHTPIN        4     // GPIO 4 for DHT11 Data pin
#define DHTTYPE       DHT11
#define MQ2PIN        36    // GPIO 36 (VP) for MQ2 Analog Output (AO)
#define BUZZER_PIN    25    // GPIO 25 for Passive/Active Buzzer
#define RED_LED_PIN   26    // GPIO 26 for Red LED (Triggered at > 200)
#define GREEN_LED_PIN 27    // GPIO 27 for Green LED (Safe at <= 200)

#define TONE_FREQ     1000  // 1 kHz sound frequency for passive buzzer

DHT dht(DHTPIN, DHTTYPE);

// --- Network & Platform Credentials ---
const char* ssid = "####"; //insert your wifi name 
const char* password = "####"; //insert your wifi password

const char* favoriot_api_key = "#######"; //find your api key
const char* device_developer_id = "########";//insert your device developer id in favoriot

// --- Favoriot MQTT Server Settings ---
const char* mqtt_server = "mqtt.favoriot.com";
const int mqtt_port = 1883;

// Topic definitions including API key prefix
String pub_topic = String(favoriot_api_key) + "/v2/streams";
String sub_topic = String(favoriot_api_key) + "/v2/rpc";

WiFiClient espClient;
PubSubClient client(espClient);

unsigned long lastMsg = 0;

// --- MQTT Callback: Listens for Dashboard Control Switch ---
void callback(char* topic, byte* payload, unsigned int length) {
  String messageBuffer;
  for (int i = 0; i < length; i++) {
    messageBuffer += (char)payload[i];
  }
  Serial.println("\n[MQTT Downlink Received]: " + messageBuffer);

  if (messageBuffer.indexOf("\"buzzer_switch\":\"ON\"") != -1 || 
      messageBuffer.indexOf("\"buzzer_switch\":1") != -1 ||
      messageBuffer.indexOf("\"BUZZ\":\"ON\"") != -1 ||
      messageBuffer.indexOf("\"BUZZ\":1") != -1) {
    
    tone(BUZZER_PIN, TONE_FREQ);
    digitalWrite(BUZZER_PIN, HIGH);
    Serial.println(">>> BUZZER ACTIVATED (ON) <<<");
  } 
  else if (messageBuffer.indexOf("\"buzzer_switch\":\"OFF\"") != -1 || 
           messageBuffer.indexOf("\"buzzer_switch\":0") != -1 ||
           messageBuffer.indexOf("\"BUZZ\":\"OFF\"") != -1 ||
           messageBuffer.indexOf("\"BUZZ\":0") != -1) {
    
    noTone(BUZZER_PIN);
    digitalWrite(BUZZER_PIN, LOW);
    Serial.println(">>> BUZZER DEACTIVATED (OFF) <<<");
  }
}

void reconnectMQTT() {
  while (!client.connected()) {
    Serial.print("Connecting to Favoriot MQTT Broker...");
    
    String clientId = "ESP32-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str(), favoriot_api_key, favoriot_api_key)) {
      Serial.println(" Connected!");
      client.subscribe(sub_topic.c_str());
      Serial.println("Subscribed to: " + sub_topic);
    } else {
      Serial.print(" Failed, rc=");
      Serial.print(client.state());
      Serial.println(" Retrying in 5 seconds...");
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  dht.begin();

  analogSetAttenuation(ADC_11db); // 0V to ~3.3V range for ADC

  pinMode(MQ2PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(RED_LED_PIN, OUTPUT);
  pinMode(GREEN_LED_PIN, OUTPUT);

  noTone(BUZZER_PIN);
  digitalWrite(BUZZER_PIN, LOW);
  digitalWrite(RED_LED_PIN, LOW);
  digitalWrite(GREEN_LED_PIN, LOW);

  WiFi.begin(ssid, password);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWi-Fi Connected!");

  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);
}

void loop() {
  if (!client.connected()) {
    reconnectMQTT();
  }
  client.loop();

  unsigned long now = millis();
  if (now - lastMsg > 30000) {
    lastMsg = now;

    float humidity = dht.readHumidity();
    float temperature = dht.readTemperature();
    int airQuality = analogRead(MQ2PIN);

    if (isnan(humidity) || isnan(temperature)) {
      Serial.println("Failed to read from DHT sensor!");
      return;
    }

    // Evaluate Threshold (> 200 triggers Dangerous state locally)
    String ledStatus;
    if (airQuality > 60) {
      digitalWrite(RED_LED_PIN, HIGH);
      digitalWrite(GREEN_LED_PIN, LOW);
      ledStatus = "RED";
      Serial.println("[ALERT] Air Quality > 200! RED LED Activated.");
    } else {
      digitalWrite(RED_LED_PIN, LOW);
      digitalWrite(GREEN_LED_PIN, HIGH);
      ledStatus = "GREEN";
      Serial.println("[OK] Air Quality <= 200. GREEN LED Activated.");
    }

    // JSON payload with temperature, humidity, air_quality, and led_status
    String jsonPayload = "{\"device_developer_id\":\"" + String(device_developer_id) + "\","
                        + "\"data\":{"
                        + "\"temperature\":" + String(temperature, 1) + ","
                        + "\"humidity\":" + String(humidity, 1) + ","
                        + "\"air_quality\":" + String(airQuality) + ","
                        + "\"led_status\":\"" + ledStatus + "\""
                        + "}}";

    if (client.connected()) {
      Serial.print("\nPublishing Telemetry: ");
      Serial.println(jsonPayload);
      
      bool success = client.publish(pub_topic.c_str(), jsonPayload.c_str());
      if (success) {
        Serial.println("--> Successfully Sent to Favoriot!");
      } else {
        Serial.println("--> Publish Failed!");
      }
    }
  }
}
